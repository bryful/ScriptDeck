cmake_minimum_required(VERSION 3.16)
project(ScriptDeck VERSION 1.0.0 LANGUAGES C CXX)
add_executable(ScriptDeck main.cpp LaunchOptions.cpp ScriptDeckApp.cpp Stack.cpp)
target_compile_features(ScriptDeck PRIVATE cxx_std_17)

add_library(scriptdeck_quickjs STATIC quickjs/quickjs.c quickjs/dtoa.c quickjs/libregexp.c quickjs/libunicode.c)
set_target_properties(scriptdeck_quickjs PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_compile_definitions(scriptdeck_quickjs PRIVATE QUICKJS_NG_BUILD _GNU_SOURCE)
if(WIN32)
    target_compile_definitions(scriptdeck_quickjs PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
endif()
if(MSVC)
    target_compile_options(scriptdeck_quickjs PRIVATE /experimental:c11atomics /W0)
else()
    target_link_libraries(scriptdeck_quickjs PUBLIC m)
endif()
add_library(scriptdeck_model STATIC ScriptModel.cpp Stack.cpp)
target_compile_features(scriptdeck_model PUBLIC cxx_std_17)
add_library(scriptdeck_script STATIC ScriptEngine.cpp ScriptIO.cpp FileDialogs.cpp)
target_compile_features(scriptdeck_script PUBLIC cxx_std_17)
target_include_directories(scriptdeck_script PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(scriptdeck_script PUBLIC scriptdeck_quickjs scriptdeck_model)
if(WIN32)
    target_link_libraries(scriptdeck_script PUBLIC ole32 shell32 uuid)
endif()
